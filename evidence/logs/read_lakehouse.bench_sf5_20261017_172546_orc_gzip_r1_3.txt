EXPLAIN ANALYZE SELECT l.orderkey, SUM(l.extendedprice*(1-l.discount)) AS revenue,
       o.orderdate, o.shippriority
FROM lakehouse.bench_sf5_20261017_172546_orc_gzip_r1.customer c
JOIN lakehouse.bench_sf5_20261017_172546_orc_gzip_r1.orders o ON c.custkey=o.custkey
JOIN lakehouse.bench_sf5_20261017_172546_orc_gzip_r1.lineitem l ON l.orderkey=o.orderkey
WHERE c.mktsegment='BUILDING' AND o.orderdate<DATE '1995-03-15'
AND l.shipdate>DATE '1995-03-15'
GROUP BY l.orderkey,o.orderdate,o.shippriority
ORDER BY revenue DESC, o.orderdate, l.orderkey LIMIT 10;
